  kubectl -n {{ .Release.Namespace }} rollout status deployment/{{ include "scarpenter.fullname" . }}
  kubectl -n {{ .Release.Namespace }} logs deployment/{{ include "scarpenter.fullname" . }}

{{- if has "none" .Values.cluster.dns }}

NOTE: cluster.dns is "none": the Pods of the nodes use the node's resolver and
cannot resolve cluster Services.
{{- end }}
{{- if eq .Values.kubeletClientCA.source "none" }}

NOTE: on k3s, kubectl logs and exec fail with 401 for the nodes unless
kubeletClientCA.source is set (the API server's client CA is not the cluster CA
there; kubeadm and EKS need nothing). See docs/how-to-guides/install.md.
{{- end }}

{{- if eq .Values.vendors.azure.credentials.mode "none" }}

{{- end }}
{{- end }}
{{- if .Values.vendors.gcp.enabled }}
{{- if eq .Values.vendors.gcp.credentials.mode "none" }}

GCP credentials are not wired by the chart (credentials.mode none): the pod must
get them from the metadata server (GKE Workload Identity, or the node's service
account) and set --gcp-project (vendors.gcp.project) if they carry no project.
{{- end }}
{{- end }}
